## Types

```ts
type SearchEntityType =
  | 'presentations'
  | 'customers'
  | 'contacts'
  | 'attendees'
  | 'business_cards'

interface SearchOptions {
  types?: SearchEntityType[]  // default: all types
  limit?: number              // max results per entity type; default 5
}

interface SearchResults {
  presentations:  { results: Presentation[]; total_count: number }
  customers:      { results: Customer[]; total_count: number }
  contacts:       { results: UserContact[]; total_count: number }
  attendees:      { results: (Attendee & { event_name: string })[]; total_count: number }
  business_cards: { results: BusinessCard[]; total_count: number }
}
```

## Query

```ts
query(text: string, options?: SearchOptions): Promise<SearchResults>
```

```js
import mobilelocker from '@mobilelocker/javascript-sdk'

const results = await mobilelocker.search.query('Acme', {
  types: ['customers', 'contacts'],
  limit: 10,
})

results.customers.results.forEach((c) => console.log(c.name))
results.contacts.results.forEach((c) => console.log(c))
// results.presentations / attendees / business_cards same shape
```

Failures throw **`MobileLockerError`**.
